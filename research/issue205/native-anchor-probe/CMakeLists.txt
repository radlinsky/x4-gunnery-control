cmake_minimum_required(VERSION 3.20)
project(x4_config_anchor_probe LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This disposable probe requires Windows x64 MSVC")
endif()
if(NOT X4NATIVE_SDK)
    message(FATAL_ERROR "Set X4NATIVE_SDK to the existing disposable host sdk directory")
endif()
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
    OUTPUT_VARIABLE probe_sha OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND git status --porcelain WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
    OUTPUT_VARIABLE probe_dirty OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(probe_dirty)
    set(dirty 1)
else()
    set(dirty 0)
endif()
add_library(x4_config_anchor_probe SHARED probe.cpp)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/probe.cpp" source_sha)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/../../../testlab/x4_gunnery_control_testlab/ui/turretholo.lua" lua_sha)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/../../../testlab/x4_gunnery_control_testlab/md/x4_gunnery_control_testlab_observe.xml" md_sha)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/../../../testlab/x4_gunnery_control_testlab/ui/scenario_spec.lua" fixture_sha)
target_compile_features(x4_config_anchor_probe PRIVATE cxx_std_17)
target_include_directories(x4_config_anchor_probe PRIVATE "${X4NATIVE_SDK}")
target_compile_definitions(x4_config_anchor_probe PRIVATE PROBE_SHA="${probe_sha}" PROBE_DIRTY=${dirty})
target_compile_definitions(x4_config_anchor_probe PRIVATE SOURCE_SHA="${source_sha}" LUA_SHA="${lua_sha}"
    MD_SHA="${md_sha}" FIXTURE_SHA="${fixture_sha}")
target_compile_options(x4_config_anchor_probe PRIVATE /W4 /WX)
target_link_libraries(x4_config_anchor_probe PRIVATE bcrypt)
set_target_properties(x4_config_anchor_probe PROPERTIES PREFIX "")
